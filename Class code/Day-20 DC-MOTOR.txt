#define IN1 4
#define IN2 3
#define ENA 5

#define POT A0
#define SWITCH A1
void setup()
{
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  pinMode(ENA, OUTPUT);
  pinMode(POT, INPUT);
  pinMode(SWITCH, INPUT_PULLUP);
}

void loop()
{
  int x= analogRead(POT);
  x= map(x,0,1023,0,255);
  bool d = digitalRead(SWITCH);
  MotorControl(d,x);
}
void MotorControl(bool Direction, byte speed)
{
  switch(Direction)
  {
    case 0: digitalWrite(IN1,1);
    		digitalWrite(IN2,0);
    		break;
    
    case 1: digitalWrite(IN1,0);
    		digitalWrite(IN2,1);
    		break;
  }
  analogWrite(ENA,speed);
}
