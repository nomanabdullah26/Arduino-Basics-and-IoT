#include <Keypad.h>

#define ROWS 4 //four rows
#define COLS 4 //three columns
#define SensorPin 4
#define LED_G 3
#define LED_R 2
#define LED_ALERT 13

bool armed_status=0, alert_status =0;
const char keys [ROWS] [COLS] = 
{
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte rowPins[ROWS] = {12,11,10,9}; //connect to the row pinouts of the kpd
byte colPins[COLS] = {8, 7, 6,5}; //connect to the column pinouts of the kpd

#define Pass_stop     "1234"
#define Pass_armed    "00"
#define Pass_unarmed  "4321"

Keypad kpd = Keypad( makeKeymap(keys), rowPins, colPins, ROWS, COLS );

void setup()
{
  Serial.begin(9600);
  pinMode(SensorPin, INPUT);
  pinMode(LED_G, OUTPUT);
  pinMode(LED_R, OUTPUT);
  pinMode(LED_ALERT, OUTPUT);
  led_check();
}
  
void readKey(String& out)
{
  static String buf="";
  char k = kpd.getKey();
  if (k)
  {
    switch(k)
    {
      case '*': buf = ""; break;
      case '#': out = buf; buf=""; break;
      default : buf= buf + String(k); break;
    }
  }
}

void pass_check()
{
  String txt;
  readKey(txt);
  if(txt.length() > 0)
  {
    if(txt == Pass_stop)
    {
      Serial.println("Alert Stopped");
      alert_status =0;
    }
    else if(txt == Pass_armed)
    {
      Serial.println("System Armed");
      armed_status=1;
    }
    else if(txt == Pass_unarmed)
    {
      armed_status=alert_status=0;
      Serial.println("System NOT Armed");
    }
    else
    {
      Serial.println("Invalid pass!");
    }
  }
}

void led_check()
{
  if(armed_status)
  {
    digitalWrite(LED_G,1);
    digitalWrite(LED_R,0);
    digitalWrite(LED_ALERT,alert_status);
  }
  else
  {
    digitalWrite(LED_G,0);
    digitalWrite(LED_R,1);
    digitalWrite(LED_ALERT,0);
  }
}

void sensor_check()
{
  bool x= digitalRead(SensorPin);
  //Serial.println(x);
  if((x==1) and (armed_status==1)) //if true
  {
    alert_status=1;
  } 
}

void loop()
{
  pass_check();
  sensor_check();
  led_check();
}
